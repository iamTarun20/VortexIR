cmake_minimum_required(VERSION 3.16)
project(VortexIR LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(search_engine engine/src/search_engine.cpp)
target_include_directories(search_engine PUBLIC engine/include)
add_executable(vortexir engine/src/main.cpp)
target_link_libraries(vortexir PRIVATE search_engine)
add_executable(engine_tests engine/tests/test_engine.cpp)
target_link_libraries(engine_tests PRIVATE search_engine)
enable_testing()
add_test(NAME engine_tests COMMAND engine_tests ${CMAKE_SOURCE_DIR}/data/resources.tsv)
